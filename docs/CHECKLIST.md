## 附录：快速验证脚本

```bash
#!/bin/bash

echo "=== 部署验证脚本 ==="

# 健康检查
echo "1. 健康检查..."
curl -s http://localhost:8080/api/health | grep -q "UP" && echo "✓ 后端正常" || echo "✗ 后端异常"
curl -s http://localhost | grep -q "html" && echo "✓ 前端正常" || echo "✗ 前端异常"

# 登录测试
echo "2. 登录测试..."
TOKEN=$(curl -s -X POST http://localhost/api/auth/login \
  -H "Content-Type: application/json" \
  -d '{"username":"admin","password":"admin123"}' | jq -r '.data.token')
echo "Token: ${TOKEN:0:20}..."

# 接口测试
echo "3. 接口测试..."
curl -s -H "Authorization: Bearer $TOKEN" http://localhost/api/system/user/list | grep -q "records" && echo "✓ 用户接口正常" || echo "✗ 用户接口异常"
curl -s -H "Authorization: Bearer $TOKEN" http://localhost/api/monitor/systems | grep -q "records" && echo "✓ 监控接口正常" || echo "✗ 监控接口异常"

echo "=== 验证完成 ==="
```
